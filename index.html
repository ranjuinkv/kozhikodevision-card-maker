🛡️ Firebase Firestore Security Rules:
To protect registered channel accounts and settings from public access, paste these rules in Firebase Console: 
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    match /kv_users/{userId} {
      allow read: if request.auth != null;
      allow write: if request.auth != null && request.auth.token.role == 'admin';
    }
    match /kv_settings/{settingId} {
      allow read: if true;
      allow write: if request.auth != null && request.auth.token.role == 'admin';
    }
  }
}
